/**
 * @param {number[]} nums
 * @return {number[][]}
 */
var subsets = function (nums) {
  const n = nums.length;

  const path = [];
  const res = [];

  const backtrack = (i) => {
    if (i === n) return res.push([...path]);
    backtrack(i + 1);

    path.push(nums[i]);
    backtrack(i + 1);
    path.pop();
  }

  backtrack(0);
  return res;
};
