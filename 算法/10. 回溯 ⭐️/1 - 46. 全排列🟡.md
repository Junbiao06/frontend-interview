/**
 * @param {number[]} nums
 * @return {number[][]}
 */
var permute = function (nums) {
  const n = nums.length;
  const path = [];
  const res = [];
  const isUsed = Array(n).fill(false);

  const backtrack = (i) => {
    if (i === n) return res.push([...path]);
    for (let j = 0; j < n; j++) {
      if (isUsed[j]) continue;

      isUsed[j] = true;
      path.push(nums[j]);

      backtrack(i + 1);

      isUsed[j] = false;
      path.pop();
    }
  }

  backtrack(0);
  return res;
};
