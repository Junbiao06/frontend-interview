/**
 * @param {string} digits
 * @return {string[]}
 */
var letterCombinations = function (digits) {
  const map = {
    '2': 'abc',
    '3': 'def',
    '4': 'ghi',
    '5': 'jkl',
    '6': 'mno',
    '7': 'pqrs',
    '8': 'tuv',
    '9': 'wxyz',
  }

  const arr = digits.split('');
  const m = arr.length;

  let path = '';
  const res = [];

  const backtrack = (i) => {
    if (i === m) return res.push(path);

    const temp = map[arr[i]].split('');
    const n = temp.length;
    for (let j = 0; j < n; j++) {
      path += temp[j];
      backtrack(i + 1);
      path = path.slice(0, -1);
    }

  }

  backtrack(0);
  return res;
};
