>  对`candiates`排序，达到剪枝效果，使得结果是升序的，不会有重复结果。

```js
/**
 * @param {number[]} candidates
 * @param {number} target
 * @return {number[][]}
 */
var combinationSum = function (candidates, target) {
  candidates.sort((a, b) => a - b);
  const n = candidates.length;

  const path = [];
  const res = [];

  const backtrack = (i, remain) => {
    if (remain === 0) return res.push([...path]);

    for (let j = i; j < n; j++) {
      if (remain - candidates[j] < 0) break;

      path.push(candidates[j]);
      backtrack(j, remain - candidates[j]);
      path.pop();
    }
  }
  backtrack(0, target);
  return res;
};
```
