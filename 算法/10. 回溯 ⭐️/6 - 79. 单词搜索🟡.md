/**
 * @param {character[][]} board
 * @param {string} word
 * @return {boolean}
 */
var exist = function (board, word) {
  const m = board.length;
  const n = board[0].length;

  const isUsed = Array.from({ length: m }, () => Array(n).fill(false));
  const isInBounds = (i, j) => i >= 0 && j >= 0 && i < m && j < n;
  const isValid = (i, j, index) => isInBounds(i, j) && !isUsed[i][j] && word[index] === board[i][j];

  const backtrack = (i, j, index) => {
    if (!isValid(i, j, index)) return false;
    if (index === word.length - 1) return true;
    isUsed[i][j] = true;
    const isFound = backtrack(i - 1, j, index + 1) || backtrack(i, j - 1, index + 1) || backtrack(i + 1, j, index + 1) || backtrack(i, j + 1, index + 1);
    isUsed[i][j] = false;
    return isFound;

  }

  for (let i = 0; i < m; i++) {
    for (let j = 0; j < n; j++) {
      if (backtrack(i, j, 0)) return true;
    }
  }
  return false;
};
