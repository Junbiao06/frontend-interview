/**
 * @param {string} s
 * @return {string[][]}
 */
var partition = function (s) {
  const isP = (s) => {
    let left = 0;
    let right = s.length - 1;

    while (left <= right) {
      if (s[left] !== s[right]) return false;
      left++;
      right--;
    }
    return true;
  }


  const path = [];
  const res = [];
  const n = s.length;

  const backtrack = (i) => {
    if (i === n) return res.push([...path]);

    for (let j = i; j < n; j++) {
      const temp = s.slice(i, j + 1);
      if (!isP(temp)) continue;
      path.push(temp);
      backtrack(j + 1);
      path.pop();
    }

  }

  backtrack(0);
  return res;


};
