/**
 * @param {number} n
 * @return {string[][]}
 */
var solveNQueens = function (n) {
  const board = Array.from({ length: n }, () => Array(n).fill('.'));
  const res = [];

  const canVisit = (row, col) => {
    for (let i = 0; i < n; i++) {
      for (let j = 0; j < n; j++) {
        if (board[i][j] === 'Q' && (i === row || j === col || i + j === row + col || i - j === row - col)) return false;
      }
    }
    return true;
  }

  const backtrack = (i) => {
    if (i === n) {
      res.push(board.map(row => row.join('')));
      return;
    }

    for (let j = 0; j < n; j++) {
      if (!canVisit(i, j)) continue;
      board[i][j] = 'Q';
      backtrack(i + 1);
      board[i][j] = '.'
    }
  }

  backtrack(0)
  return res;
};
