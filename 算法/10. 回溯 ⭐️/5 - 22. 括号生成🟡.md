/**
 * @param {number} n
 * @return {string[]}
 */
var generateParenthesis = function (n) {
  let path = '';
  const res = [];

  const backtrack = (left, right) => {
    if (left + right === 2 * n) return res.push(path);
    if (left < n) {
      path += '(';
      backtrack(left + 1, right);
      path = path.slice(0, -1);
    }

    if (right < left) {
      path += ')';
      backtrack(left, right + 1);
      path = path.slice(0, -1);
    }
  }

  backtrack(0, 0);
  return res;
};
