/**
 * Definition for singly-linked list.
 * function ListNode(val, next) {
 *     this.val = (val===undefined ? 0 : val)
 *     this.next = (next===undefined ? null : next)
 * }
 */
/**
 * @param {ListNode} head
 * @param {number} k
 * @return {ListNode}
 */
var reverseKGroup = function (head, k) {
  const getKGroup = (head, k) => {
    let tail = head;

    for (let i = 0; i < k - 1; i++) {
      if (!tail) return null;
      tail = tail.next;
    }
    return tail ? [head, tail] : null;
  }

  const reverseGroup = (head, tail) => {
    let curr = head;
    let prev = null;
    let stop = tail.next;

    while (curr !== stop) {
      let next = curr.next;
      curr.next = prev;
      prev = curr;
      curr = next;
    }

    return [tail, head];
  }

  const dummy = new ListNode(null, head);
  let prev = dummy;

  while (prev.next) {
    const group = getKGroup(prev.next, k);
    if (!group) break;

    let [start, end] = group;
    let next = end.next;
    [start, end] = reverseGroup(start, end);
    prev.next = start;
    end.next = next;

    prev = end;
  }

  return dummy.next;
};
