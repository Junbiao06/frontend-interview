## 1. 计算长度
> 为了方便删除第一个节点，引入`dummy`头节点，并且注意不要直接返回`head`。
```js
/**
 * Definition for singly-linked list.
 * function ListNode(val, next) {
 *     this.val = (val===undefined ? 0 : val)
 *     this.next = (next===undefined ? null : next)
 * }
 */
/**
 * @param {ListNode} head
 * @param {number} n
 * @return {ListNode}
 */
var removeNthFromEnd = function (head, n) {
  const dummy = new ListNode(null, head);

  let p = head;
  let length = 0;
  while (p) {
    p = p.next;
    length++;
  }

  p = dummy;
  for (let i = 0; i < length - n; i++) {
    p = p.next;
  }
  p.next = p.next.next;

  // 不能用head，因为 head 可能被删除
  return dummy.next;
};
```

## 2. 快慢指针
> 为了方便删除第一个节点，引入`dummy`头节点，并且注意不要直接返回`head`。
> `fast`先走`n`步，然后两个指针一起走，可以控制`slow`到目标的前一个节点。

```js
/**
 * Definition for singly-linked list.
 * function ListNode(val, next) {
 *     this.val = (val===undefined ? 0 : val)
 *     this.next = (next===undefined ? null : next)
 * }
 */
/**
 * @param {ListNode} head
 * @param {number} n
 * @return {ListNode}
 */
var removeNthFromEnd = function (head, n) {
  const dummy = new ListNode(null, head);

  let fast = dummy;
  let slow = dummy;

  for (let i = 0; i < n; i++) {
    fast = fast.next;
  }

  while (fast.next) {
    fast = fast.next;
    slow = slow.next;
  }

  slow.next = slow.next.next;
  // 不能用head，因为 head 可能被删除
  return dummy.next;
};
```
