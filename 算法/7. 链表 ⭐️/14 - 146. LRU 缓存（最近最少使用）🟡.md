class MyListNode {
  key: number;
  value: number;
  prev: MyListNode | null;
  next: MyListNode | null;

  constructor(key: number = 0, value: number = 0) {
    this.key = key;
    this.value = value;
    this.prev = null;
    this.next = null;
  }
}

class LRUCache {
  capacity: number;
  cache: Map<number, MyListNode>;
  head: MyListNode;
  tail: MyListNode;


  constructor(capacity: number) {
    this.capacity = capacity;
    this.cache = new Map();
    this.head = new MyListNode();
    this.tail = new MyListNode();
    this.head.prev = null;
    this.head.next = this.tail; // 指向最旧节点
    this.tail.prev = this.head; // 指向最新节点
    this.tail.next = null;
  }

  get(key: number): number {
    const cache = this.cache;

    if (cache.has(key)) {
      const node = cache.get(key);
      this.updateNode(node)
      return node.value;
    } else {
      return -1;
    }
  }

  put(key: number, value: number): void {
    const head = this.head;
    const cache = this.cache;
    const capacity = this.capacity;

    const node = new MyListNode(key, value)

    if (cache.has(key)) {
      this.deleteNode(cache.get(key));
    }
    this.addNode(node);

    cache.set(key, node);

    if (cache.size > capacity) {
      const oldest = head.next!;
      this.deleteNode(oldest)
      cache.delete(oldest.key);
    }
  }

  addNode(node: MyListNode): void {
    const tail = this.tail;
    const prev = tail.prev;

    node.prev = prev;
    node.next = tail;

    tail.prev = node;
    prev.next = node;
  }

  deleteNode(node: MyListNode): void {
    const prev = node.prev!;
    const next = node.next!;

    prev.next = node.next;
    next.prev = node.prev;

    node.prev = null;
    node.next = null;
  }

  updateNode(node: MyListNode): void {
    this.deleteNode(node);
    this.addNode(node);
  }
}

/**
 * Your LRUCache object will be instantiated and called as such:
 * var obj = new LRUCache(capacity)
 * var param_1 = obj.get(key)
 * obj.put(key,value)
 */
